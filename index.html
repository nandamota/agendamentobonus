// ============================================= // SLOTS DISPONÍVEIS — edite aqui para atualizar // Formato: 'AAAA-MM-DD': número de vagas // ============================================= const SLOTS_POR_DATA = { '2026-04-15': ['07:00','07:30','08:00','08:30','09:00','10:00','10:30','11:00','11:30'], '2026-04-17': ['10:00','10:30','11:00','11:30'], '2026-04-22': ['07:00','07:30','08:00','08:30','09:00','10:00','10:30','11:00','11:30'], '2026-04-24': ['10:00','10:30','11:00','11:30'], '2026-04-29': ['07:00','07:30','08:00','08:30','09:00','09:30','10:30','11:00','11:30'], '2026-05-01': ['10:00','10:30','11:00','11:30'], '2026-05-06': ['07:00','07:30','08:00','08:30','09:00','09:30','10:30','11:00','11:30'], }; // ============================================= // NÃO PRECISA ALTERAR ABAIXO DESTA LINHA // ============================================= function doGet() { try { const agora = new Date(); const hoje = new Date(agora.getFullYear(), agora.getMonth(), agora.getDate()); const diasNomesCompletos = ['Domingo','Segunda-feira','Terça-feira','Quarta-feira','Quinta-feira','Sexta-feira','Sábado']; const meses = ['jan','fev','mar','abr','mai','jun','jul','ago','set','out','nov','dez']; const resultado = []; for (const dataStr in SLOTS_POR_DATA) { const [ano, mes, dia] = dataStr.split('-').map(Number); const diaDate = new Date(ano, mes - 1, dia); if (diaDate < hoje) continue; const slots = SLOTS_POR_DATA[dataStr]; resultado.push({ data: dia + ' ' + meses[mes - 1], diaSemana: diasNomesCompletos[diaDate.getDay()], vagas: slots.length, horarios: slots }); } const output = ContentService.createTextOutput(JSON.stringify(resultado)); output.setMimeType(ContentService.MimeType.JSON); return output; } catch (e) { const erro = ContentService.createTextOutput(JSON.stringify({ erro: e.message })); erro.setMimeType(ContentService.MimeType.JSON); return erro; } }
